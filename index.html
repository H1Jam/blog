This helper keeps WordPress post dates human friendly when rendering the feed.

function formatDate(dateString) {
  return new Date(dateString).toLocaleDateString('en-US', {
    year: 'numeric',
    month: 'long',
    day: 'numeric'
  });
}

console.log(formatDate('2021-04-02T13:53:23'));
// ➜ April 2, 2021
